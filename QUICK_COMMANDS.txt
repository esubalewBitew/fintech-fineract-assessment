Tomorrow quick commands — fineract-assessment
===========================================

cd Fineract-Dev-Project/fineract-assessment

# Stack
docker compose ps
docker compose logs fineract --tail 50
docker compose up -d

# Health
curl -s http://localhost:8081/health
curl -s http://localhost:8082/health
curl -sk -o /dev/null -w '%{http_code}\n' https://localhost:8443/fineract-provider/actuator/health

# CoA sync (A-sync)
curl -s -X POST http://localhost:8082/sync/coa \
  -H 'X-Api-Key: integration-key' | jq .

# Part B setup (after CoA)
. .venv/bin/activate
pip install -r scripts/requirements.txt
PYTHONPATH=scripts python scripts/fineract_setup_part_b.py --out evidence/01-part-b-setup.json

# Fineract checks (bootstrap admin)
curl -sk -u mifos:password -H 'Fineract-Platform-TenantId: default' \
  https://localhost:8443/fineract-provider/api/v1/offices | jq .
curl -sk -u mifos:password -H 'Fineract-Platform-TenantId: default' \
  https://localhost:8443/fineract-provider/api/v1/loanproducts | jq .
curl -sk -u mifos:password -H 'Fineract-Platform-TenantId: default' \
  https://localhost:8443/fineract-provider/api/v1/currencies | jq '.selectedCurrencyOptions'

# Mock ERP
curl -s http://localhost:8081/accounts -H 'X-Api-Key: mock-erp-key' | jq '.[].external_reference'
curl -s 'http://localhost:8081/journal-entries' -H 'X-Api-Key: mock-erp-key' | jq .

# Map loan then money (needs real fineract_loan_id)
curl -s -X POST http://localhost:8082/loans/map \
  -H 'X-Api-Key: integration-key' -H 'Content-Type: application/json' \
  -d '{"loan_reference":"LOAN-DEMO-1","fineract_loan_id":REPLACE_ME}'

curl -s -X POST http://localhost:8082/loans/disburse \
  -H 'X-Api-Key: integration-key' -H 'Content-Type: application/json' \
  -d '{"external_reference":"DISB-1","loan_reference":"LOAN-DEMO-1","amount":100000,"currency":"RWF"}'

curl -s -X POST http://localhost:8082/loans/repay \
  -H 'X-Api-Key: integration-key' -H 'Content-Type: application/json' \
  -d '{"external_reference":"REPAY-1","loan_reference":"LOAN-DEMO-1","amount":10000,"currency":"RWF","principal":8000,"interest":2000}'

curl -s 'http://localhost:8082/reconcile/fineract-erp' -H 'X-Api-Key: integration-key' | jq .
curl -s 'http://localhost:8082/admin/recover' -H 'X-Api-Key: integration-key' | jq .

# Tests
. .venv/bin/activate
PYTHONPATH=scripts python -m pytest scripts/tests -q
cd integration-service && PYTHONPATH=. python -m pytest -q
cd ../mock-erp && PYTHONPATH=. python -m pytest -q

# DB owners check
docker compose exec db psql -U root -c '\du' -c '\l mock_erp' -c '\l integration'
